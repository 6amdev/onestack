# 1. Clone หรือ download OneStack
cd ~
git clone <onestack-repo>
cd onestack

# 2. สร้าง config
cp config.example.yml config.yml
nano config.yml

# 3. ติดตั้ง
sudo bash install.sh

# 4. ทดสอบ
sudo bash test-install.sh

# 5. ดู credentials
cat /opt/onestack/.credentials

# 6. เข้าใช้งาน
# เปิด browser ไปที่ http://localhost หรือ http://your-domain
